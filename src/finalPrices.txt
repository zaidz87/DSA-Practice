class Solution {
public:
    vector<int> finalPrices(vector<int>& prices) {
        stack<int> s;
        vector<int> ans = prices;
        for(int i = 0 ; i < prices.size();i++){

            while(!s.empty() && prices[i] <= prices[s.top()]){
                int index = s.top();
                ans[index] = prices[index] - prices[i];
                s.pop();

            }
            s.push(i);
        }
        return ans;
    }
};
